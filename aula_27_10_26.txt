DELIMITER $$
CREATE FUNCTION obternomesclientes ()
RETURNS VARCHAR(255)
BEGIN
    DECLARE nomes VARCHAR(255);
    SELECT GROUP_CONCAT ( nome SEPARATOR ',') INTO nomes FROM clientes;
    RETURN nomes;
END $$

STORED PROCEDURE

DELIMITER $$
CREATE PROCEDURE InserirCliente(
    IN p_nome VARCHAR(255),
    IN p_email VARCHAR(255),
    IN p_telefone VARCHAR(20)
)
BEGIN
    INSERT INTO clientes (nome,email,telefone)
    VALUES (p_nome, p_email, p_telefone);
END $$
DELIMITER;
